// Node.js 18+ · npm i ws (Node 22'de gerekmez)
const { Kanka } = require('./kanka.js');
const bot = new Kanka({ token: process.env.KANKA_TOKEN, durum: { durum: 'cevrimici', aktivite: { tur: 'oynuyor', ad: '/zar' } } });

bot.komut('zar', { aciklama: 'Zar at', secenekler: [{ ad: 'yuz', tur: 'tamsayi', aciklama: 'Kaç yüzlü?' }] },
  k => k.cevapla(`🎲 ${1 + Math.floor(Math.random() * k.secenek('yuz', 6))}`));

bot.on('mesaj', m => { if (m.icerik === 'ping') m.cevapla('pong 🏓'); });
bot.on('hazir', () => console.log(bot.ben.ad, 'hazır!'));
bot.baglan();


# Python 3.9+ · pip install aiohttp
import os, random, kanka
bot = kanka.Bot(token=os.environ["KANKA_TOKEN"], durum={"durum": "cevrimici", "aktivite": {"tur": "oynuyor", "ad": "/zar"}})

@bot.komut("zar", aciklama="Zar at", secenekler=[kanka.Secenek("yuz", "tamsayi", "Kaç yüzlü?")])
async def zar(k):
    await k.cevapla(f"🎲 {random.randint(1, k.secenek('yuz', 6))}")

@bot.olay("mesaj")
async def mesaj(m):
    if m["icerik"] == "ping": await m.cevapla("pong 🏓")

bot.calistir()

    
Token botunun şifresidir. Koda yazma, GitHub'a yükleme. Sızarsa portaldan Token'ı yenile; eskisi anında çalışmaz olur.

    
Botunu sunucuya ekletme

    
Botu bir sunucuya sadece o sunucunun yöneticisi ekleyebilir. Senin işin ona doğru ekleme linkini vermek. Link şu şekildedir:

    
https://kankahq.com/bot-ekle/<botun-uygulama-kimliği>?izin=manageMessages,kick,timeout

    
Linki nereden alırım? Üç yol var, üçü de aynı linki verir:

    
YolNasıl

      
PortalGeliştirici Portalı › botun › Sunucuya ekleme linki. Kutulardan botun gerçekten kullandığı izinleri işaretle, link kendiliğinden güncellenir; kopyala.

      
Koddan (kanka.js 1.4.1+)bot.eklemeLinki(['manageMessages', 'kick']): .env dosyasına link yazmana gerek kalmaz.

      
Koddan (kanka.py 1.4.1+)bot.ekleme_linki("manageMessages", "kick")

    

    
Botuna bir /davet komutu koyup linki oradan vermek en kolayı:

    
bot.komut('davet', { aciklama: 'Botu kendi sunucuna ekle' }, async k => {
  await k.cevapla({ kartlar: [new Kart().baslik('Beni sunucuna ekle 🤖')
    .aciklama('Butona bas, sunucunu seç, izinleri onayla. Sunucuda "Sunucuyu yönet" yetkin olmalı.')],
    bilesenler: [[{ tur: 'buton', etiket: 'Sunucuna ekle', url: bot.eklemeLinki(['manageMessages', 'kick', 'timeout']) }]] });
});

    
Linki açan kişi ne görür? Botun adı ve açıklaması, yönetebildiği sunucuların listesi ve botun istediği izinler. Sunucuyu seçip Sunucuya ekle'ye basar. Bot o anda sunucuya kendi adını taşıyan bir rolle girer. İzinler o role verilir; yönetici bunları sonra Sunucu ayarları › Roller'den değiştirebilir. Bot girince sana SUNUCUYA_EKLENDI olayı gelir (kanka.js: bot.on('sunucuyaEklendi', s => …)); karşılama mesajını burada atabilirsin. Bot atılırsa SUNUCUDAN_CIKARILDI gelir.

    
Hangi izinleri istemeliyim? Sadece kullandıklarını. Uzun bir izin listesi yöneticiyi korkutur ve botun eklenmez. Mesaj göndermek, komutlara cevap vermek ve mesajları görmek için izin gerekmez.

2. Komutlar

    
Komutlar kullanıcıların / listesinde botun adı ve simgesiyle çıkar. Kanka'nın kendi komutuyla aynı adı taşıyorsa /ad:botadi olarak görünür. Seçenekleri olan bir komutu kullanıcı listeden seçince Kanka seçenekler için bir form açar; /zar 20 ya da /zar yuz:20 gibi yazmak da olur.

    
Seçenek türüBota gelen değer

      
yazimetin (en fazla 1000 karakter)

      
sayi / tamsayisayı

      
evethayirtrue / false

      
kisikullanıcı adı (sunucuda olmalı)

      
oda / roloda ya da rol kimliği

    
Sabit seçimler için secimler: [{ ad: 'Yazı', deger: 'yazi' }] ekle. Bir komutu tek sunucuya özel yapmak için JavaScript'te { sunucu: '<id>' }, Python'da sunucu="<id>". En fazla 50 komut, komut başına 10 seçenek.

    
3 saniye kuralı: komuta 3 saniye içinde cevap vermelisin. Uzun sürecek işler için önce k.bekle() de (kullanıcı "düşünüyor…" görür), sonra 15 dakika içinde k.duzenle('...') ile bitir. k.cevapla({ icerik, gizli: true }) cevabı sadece komutu kullanan kişiye gösterir.


bot.komut('rapor', { aciklama: 'Uzun rapor' }, async k => {
  await k.bekle();
  const sonuc = await uzunIs();
  await k.duzenle(`Hazır: ${sonuc}`);
});


    
3. Butonlar ve seçim menüleri

    
Bir mesaja en fazla 5 satır ekleyebilirsin; bir satırda 5 buton ya da 1 seçim menüsü olur. Buton stilleri: ana, ikincil, basari, tehlike, link (link butonu url açar, bota olay gelmez).

bot.komut('oylama', { aciklama: 'Hızlı oylama' }, k => k.cevapla({

  bilesenler: [

  ],

bot.buton(/^(evet|hayir)$/, e => e.guncelle(`${e.kullanici.gorunenAd}: ${e.veri.id}`)); // mesajı yerinde değiştirir

Üye, rol ve oda seçici

    
İsim yazdırmak yerine kullanıcıya listeden seçtir: kişi sunucudaki üyeler arasında arar, tıklar. Yanlış yazılmış isim, olmayan kişi derdi yok. Liste Kanka'dan gelir, botun seçenek göndermesine gerek yok. Ticket, etkinlik ve rol botları için birebir.


const { KisiSecici, RolSecici, OdaSecici } = require('./kanka.js');
k.cevapla({ icerik: 'Bilete kimi ekleyelim?', bilesenler: [[KisiSecici('ekle', { max: 3, yerTutucu: 'Üye ara…' })]] });
bot.secim('ekle', async e => {
  for (const u of e.secilenler.kullanicilar) await bot.odaIzni(e.oda, u.ad, { view: true, send: true });
  e.cevapla({ icerik: `${e.degerler.join(', ')} bilete eklendi`, gizli: true });
});
// RolSecici('rol') → e.secilenler.roller · OdaSecici('oda', { odaTuru: 'voice' }) → e.secilenler.odalar

    
Seçilen değerler e.degerler'de (kişi adı, rol id'si, oda id'si), tam bilgileri e.secilenler'de gelir (üyenin rolleri, izinleri, Energy'si). Kanka seçimi kendisi doğrular: sunucuda olmayan biri seçilemez. Formlarda da kullanılır: alan tur: 'kisi' (ya da 'rol', 'oda') olunca formda arama kutusu çıkar, sonucu e.secilenler.alanId.


    
Sadece bir kişinin gördüğü mesaj

    
e.gizliCevapla() 15 dakika sonra kaybolur. Kalıcı olsun istersen (ticket paneli, kişiye özel bilgi): e.kaliciGizliCevapla(icerik) ya da etkileşim olmadan bot.ozelGonder(oda, 'ali', icerik). Mesaj odada sadece o kişiye görünür, sayfa yenilense de kalır; kişi "Kapat" diyene ya da bot bot.ozelSil(oda, id) diyene kadar durur. Üzerindeki butonlar ve seçiciler çalışır. Bir odada kişi başı en çok 30 tane tutulur, eskisi silinir.


    
Kişinin yetkisini kontrol etme

    
Etkileşimde basan kişinin sunucu yetkileri zaten gelir: e.yetkisiVar('manageMessages') (yönetici ve sahip her zaman evet). Etkileşim dışında: await bot.yetkisiVar(sunucu, 'ali', 'kick'), odaya özel izin için await bot.yetkisiVar(sunucu, 'ali', 'send', oda). Ayrıntılı hâli bot.yetkiler(sunucu, 'ali', oda): { sahip, yonetici, izinler, roller, oda: { view, send, ... } }. Rollerin izinleri bot.roller(sunucu)'da, botu sunucuya kimin eklediği bot.sunucu(id).ekleyen'de.

    
Bot "yetki doğrulanamadı" diyorsa: bota ne yapacaksa o izni sunucuya eklerken vermek gerekir (ekleme linkindeki kutular) ve botun rolü, işlem yapacağı kişinin en yüksek rolünden üstte olmalı (Sunucu ayarları › Roller'de sürükle). bot.yetkiler() cevabındaki botunUstunde: true bu sorunu gösterir. Sunucu sahibine bot hiçbir şey yapamaz.

Botun durumu ve etkinliği

    
Botun yanındaki nokta (çevrimiçi / boşta / rahatsız etmeyin / görünmez) ve altındaki yazı ("Oynuyor: Minecraft") istediğin zaman değişir. En kolayı botu oluştururken vermek: bağlanınca kendiliğinden ayarlanır, bağlantı kopup gelse de korunur.

    


const bot = new Kanka({ token, durum: { durum: 'rahatsizetme', aktivite: { tur: 'dinliyor', ad: 'Spotify' } } });

await bot.durum('bosta');                          // sadece nokta
await bot.durum('cevrimici', 'Yardım için /yardim'); // nokta + serbest yazı
await bot.aktivite('oynuyor', 'Minecraft');        // "Oynuyor: Minecraft"
await bot.durum({ durum: 'gorunmez' });            // çevrimdışı görün
console.log(await bot.durumuGetir());              // { durum, yazi, bagli }

Etkinlik türleri: oynuyor, dinliyor, izliyor, yayinda (ya da yayın, canlı), yarisiyor, ozel (önüne bir şey eklenmez). Bilinmeyen tür artık hata verir. Yazı en fazla 80 karakter; durum dakikada en fazla 10 kez değişir. Durum ve yazı, bot ağ geçidine bağlıyken görünür; bağlı değilken bot çevrimdışı görünür.

    
Kütüphane kullanmıyorsan: PATCH /api/v1/ben/durum ile { "durum": "bosta", "aktivite": { "tur": "izliyor", "ad": "YouTube" } }, ya da ağ geçidinde { "op": 3, "d": { ... } } gönder (cevap olarak DURUM_GUNCELLENDI gelir). Tanıtırken (op 2) d.durum da verebilirsin. Discord'daki { status: 'dnd', activities: [{ name, type }] } biçimi de kabul edilir.

8. Gateway (kütüphane kullanmayanlar için)

    
wss://kankahq.com/bot/gateway adresine bağlan. Her mesaj { op, d, t, s } biçiminde JSON'dur.

    
opAnlamı

      
10 MERHABASunucu gönderir: { nabizAraligi: 30000 }

      
2 TANITSen gönderirsin: { token, niyetler: [...], durum? }

      
3 DURUMSen gönderirsin: { durum, yazi, aktivite }; cevap DURUM_GUNCELLENDI ya da HATA

      
1 NABIZ / 11 NABIZ_ALINDIHer nabizAraligi ms'de bir 1 gönder; 75 saniye gelmezse bağlantı kapanır

      
0 OLAYt: HAZIR, MESAJ_GELDI, MESAJ_DUZENLENDI, MESAJ_SILINDI, KOMUT, BUTON, SECIM, FORM, UYE_KATILDI, UYE_AYRILDI, UYE_GUNCELLENDI, TEPKI_EKLENDI, TEPKI_KALDIRILDI, SUNUCUYA_EKLENDI, SUNUCUDAN_CIKARILDI, ODA_OLUSTU, ODA_GUNCELLENDI, ODA_SILINDI; s sıra numarası

      
6 DEVAMKoptuktan sonra 2 dakika içinde: { token, oturum, sira }; kaçırdığın olaylar gelir. Sunucu güncellemesinden sonra da çalışır

      
7 YENİDEN_BAĞLANSunucu gönderir: { bekle: 4000 }. Sunucu birkaç saniyeliğine güncelleniyor; bekle ms sonra yeniden bağlan ve 6 DEVAM ile aynı oturuma dön

    
Güncellemede kopma: Kanka güncellenirken bağlantı 1012 koduyla kapanır ve REST birkaç saniye 502/503 dönebilir. Bağlantı kopunca her zaman yeniden bağlan (bekleme 1 sn'den başlayıp en çok 15 sn'ye çıksın), REST'te 502/503/504 gelirse 1-2 sn bekleyip tekrar dene. kanka.js ve kanka.py (1.2.0 ve sonrası) bunların hepsini kendisi yapar; eski sürümü kullanıyorsan güncelle.

    
Mesajı kim yazdı? Her mesajda yazar alanı vardır: { ad, gorunenAd, bot }. Yazar bir geliştirici botuysa yazar.uygulama da gelir: { id, ad, gorunenAd, dogrulandi, sahip }. Kanka'nın kendi botunun mesajlarında yazar.sistem: true olur. Botların birbirine cevap verip döngüye girmemesi için yazar.bot doğruysa mesajı yok saymanı öneririz.

    
Kapanış kodları: 4004 token geçersiz, 4005 zaten 2 bağlantı var, 4008 çok sık, 4014 bot durduruldu (4004 ve 4014 dışında her kapanışta yeniden bağlan), 1012 sunucu güncelleniyor.

10. Sınırlar ve kurallar

    
    
      
  • Erişim sınırı: doğrulanmamış bir bot en fazla 50 farklı kişiye ulaşabilir (botun olduğu sunuculardaki farklı insanlar). Bu sınırı geçecek bir sunucuya eklenemez. 2 sunucuda 50 kişi de olsa, 5 sunucuda 30 kişi de olsa sayılan, kaç farklı kişiye ulaştığıdır. Portaldan doğrulama iste; Kanka ekibi botu inceleyip onaylarsa sınır kalkar ve adının yanında tik çıkar.

  • Hız: bot başına saniyede 10 istek; oda başına 5 saniyede 10 mesaj; dakikada 30 DM. Aşarsan 429 ve tekrarDene (saniye) döner. Sürekli aşan bot 1 saat durur.
    • 
      
  • Özel mesaj: bot sadece ortak bir sunucusu olan ve "Botlar bana özel mesaj atabilsin" ayarını kapatmamış kişilere yazabilir. Bot DM'leri uçtan uca şifreli değildir; kullanıcının kendi şifreli DM'lerini botlar asla göremez.

  • Bir sunucuda en fazla 25 bot olur; herkese açık sunuculara bot eklenmez.
